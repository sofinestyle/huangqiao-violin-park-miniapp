import {readFileSync,writeFileSync} from 'node:fs';import assert from 'node:assert/strict';import {DatabaseSync} from 'node:sqlite';import {createHash} from 'node:crypto';
const root='/Users/aaron/Documents/huangqiao-app',out=root+'/docs/design/ui-tour-complete-2026-10-02',base='http://127.0.0.1:8787';
const credentials=readFileSync(root+'/.local/admin-access.txt','utf8').split('\n').filter(x=>x.includes('：')).map(x=>x.slice(x.indexOf('：')+1).trim());
const login=await fetch(base+'/api/admin/login',{method:'POST',headers:{'Content-Type':'application/json','X-HQ-Action':'1'},body:JSON.stringify({username:credentials[0],password:credentials[1]})});
assert.ok(login.ok,'Existing local admin sign-in failed');const cookie=login.headers.get('set-cookie').split(';')[0];
const get=async path=>{const r=await fetch(base+path,{headers:{Cookie:cookie}});assert.ok(r.ok,'Read failed');return r.json()};
const samples=JSON.parse(readFileSync(out+'/approved-content.json','utf8'));const before=await get('/api/admin/content?kind=spot');writeFileSync(out+'/spots-before.json',JSON.stringify(before,null,2));const results=[];
for(const target of samples){const previous=before.find(x=>x.id===target.id);assert.equal(previous.kind,'spot');assert.equal(previous.isTest,true);const r=await fetch(base+'/api/admin/content/'+target.id,{method:'PUT',headers:{Cookie:cookie,'Content-Type':'application/json','X-HQ-Action':'1'},body:JSON.stringify({name:previous.name,state:previous.state,sort:previous.sort,version:previous.version,data:{...previous,description:target.description,tags:target.tags,visitItems:target.visitItems}})});assert.ok(r.ok,'Content update failed: '+target.id+' '+r.status);const after=await r.json();for(const key of ['id','kind','name','images','opening','visitNote','state','sort','isTest'])assert.deepEqual(after[key],previous[key]);results.push({id:target.id,versionBefore:previous.version,versionAfter:after.version,tagCount:after.tags.length,itemCount:after.visitItems.length});}
const after=await fetch(base+'/api/public/content?kind=spot').then(r=>r.json());for(const x of samples){const row=after.find(y=>y.id===x.id);for(const key of ['description','tags','visitItems'])assert.deepEqual(row[key],x[key]);}writeFileSync(out+'/spots-after.json',JSON.stringify(after,null,2));
await fetch(base+'/api/admin/logout',{method:'POST',headers:{Cookie:cookie,'Content-Type':'application/json','X-HQ-Action':'1'},body:'{}'});
const snapshot=new DatabaseSync(root+'/backups/2026-10-02-CR006-before/huangqiao.sqlite',{readOnly:true}),db=new DatabaseSync(root+'/.local/huangqiao.sqlite',{readOnly:true});const checks={};for(const t of ['bookings','consultations','slots','changes','accounts','visitors','media','migrations','idempotency']){const rows=d=>d.prepare(`SELECT * FROM ${t} ORDER BY rowid`).all();checks[t]=JSON.stringify(rows(snapshot))===JSON.stringify(rows(db));assert.equal(checks[t],true,t+' changed');}
const previousRows=snapshot.prepare('SELECT * FROM content ORDER BY id').all();for(const row of previousRows){if(samples.some(x=>x.id===row.id))continue;assert.deepEqual(db.prepare('SELECT * FROM content WHERE id=?').get(row.id),row);}
const report={date:new Date().toISOString(),method:'existing admin API login and PUT with permission/version/audit',backup:'backups/2026-10-02-CR006-before',updates:results,otherContentRecordsUnchanged:previousRows.length-samples.length,protectedTablesUnchanged:checks,schemaUnchanged:JSON.stringify(snapshot.prepare("SELECT name,sql FROM sqlite_master WHERE type='table' ORDER BY name").all())===JSON.stringify(db.prepare("SELECT name,sql FROM sqlite_master WHERE type='table' ORDER BY name").all()),auditContentUpdates:db.prepare("SELECT COUNT(*) AS n FROM audit WHERE id>? AND action='content.update'").get(snapshot.prepare('SELECT MAX(id) AS n FROM audit').get().n).n};writeFileSync(out+'/content-restoration.json',JSON.stringify(report,null,2));db.close();snapshot.close();console.log(JSON.stringify(report));
